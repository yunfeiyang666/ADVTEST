```python
def calculate_relative_position(ego_pose, obj_annotation):
    """
    计算相对位置（最重要的函数！）
    
    转换：全局坐标 → ego坐标系
    
    例子：
    ego在 [411, 1181, 0]，朝向北
    car在 [354, 1132, 1]
    
    → 全局相对位置：[-57, -49, 1]
    → ego坐标系：[65, -37, 1]（前方65米，右侧37米）
    """
    
    # 1. 全局相对位置
    rel_pos_global = obj_translation - ego_translation
    
    # 2. 提取ego朝向角
    ego_yaw = quaternion_to_yaw(ego_rotation)
    
    # 3. 旋转变换
    cos_yaw = np.cos(-ego_yaw)
    sin_yaw = np.sin(-ego_yaw)
    
    rel_x = cos_yaw * rel_pos_global[0] - sin_yaw * rel_pos_global[1]
    rel_y = sin_yaw * rel_pos_global[0] + cos_yaw * rel_pos_global[1]
    
    # 4. 计算角度
    rel_angle = np.arctan2(rel_y, rel_x) * 180 / np.pi
    
    return rel_pos, rel_angle
```

### **关键函数2：evaluate_spatial_predicates()**

```python
def evaluate_spatial_predicates(rel_pos, rel_angle, velocity):
    """
    评估S3C空间谓词（核心转换！）
    
    转换：连续数值 → 离散谓词
    
    例子：
    distance = 75.25米 → 超出25米 → 无等级
    angle = -29.6度 → "front"
    speed = 0.04 m/s → "stopped"
    """
    predicates = []
    
    # 1. 距离谓词（5档）
    distance = np.linalg.norm(rel_pos[:2])
    
    if distance <= 4:
        predicates.append('near_coll')  # 0-4米
    elif distance <= 7:
        predicates.append('super_near')  # 4-7米
    elif distance <= 10:
        predicates.append('very_near')  # 7-10米
    elif distance <= 16:
        predicates.append('near')  # 10-16米
    elif distance <= 25:
        predicates.append('visible')  # 16-25米
    # 超过25米：无距离等级
    
    # 2. 方向谓词（4档）
    angle = rel_angle
    
    if -45 <= angle < 45:
        predicates.append('front')  # 前方
    elif 45 <= angle < 135:
        predicates.append('left')  # 左侧
    elif 135 <= abs(angle) <= 180:
        predicates.append('rear')  # 后方
    else:
        predicates.append('right')  # 右侧
    
    # 3. 运动谓词（2种）
    speed = np.linalg.norm(velocity[:2])
    
    if speed > 0.5:
        predicates.append('moving')  # 移动
    else:
        predicates.append('stopped')  # 静止
    
    return predicates
```
